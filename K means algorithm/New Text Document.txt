import numpy as np
import pandas as pd
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
import matplotlib.pyplot as plt

# -------------------------------------------------------
# LOAD WINE DATASET FROM LOCAL CSV (not from UCI URL)
# -------------------------------------------------------
# Your local CSV must contain the same 14 columns.
file_path = "wine.csv"   # <--- replace with your file name

column_names = [
    'Class', 'Alcohol', 'Malic_Acid', 'Ash', 'Alcalinity_of_Ash', 'Magnesium',
    'Total_Phenols', 'Flavanoids', 'Nonflavanoid_phenols', 'Proanthocyanins',
    'Color_Intensity', 'Hue', 'OD280/OD315_of_diluted_wine', 'Proline'
]

wine_data = pd.read_csv(file_path, header=None, names=column_names)
# -------------------------------------------------------

# Step 1: Standardize features
features = wine_data.drop('Class', axis=1)
scaler = StandardScaler()
features_scaled = scaler.fit_transform(features)

# Step 2: Apply K-Means clustering
kmeans = KMeans(n_clusters=3, random_state=42)
wine_data['Cluster'] = kmeans.fit_predict(features_scaled)

# Step 3: Visualization using 2 features
plt.figure(figsize=(8, 6))
plt.scatter(
    wine_data['Alcohol'],
    wine_data['Malic_Acid'],
    c=wine_data['Cluster'],
    cmap='viridis'
)
plt.xlabel('Alcohol')
plt.ylabel('Malic Acid')
plt.title('K-Means Clustering on Wine Dataset (CSV)')
plt.colorbar(label='Cluster')
plt.show()

# Step 4: Show first few rows
print(wine_data.head())
