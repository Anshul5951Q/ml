"""
bank_knn.py

KNN classifier pipeline for the "bank-full" dataset (bank marketing).
Assumptions:
 - CSV file: "bank-full.csv" in the working directory
 - Target column: "y" with values 'yes' or 'no'
 - The "duration" column (if present) should be removed because it leaks information.

Requirements:
 pip install pandas numpy scikit-learn imbalanced-learn joblib

This script:
 - loads data
 - does cleaning and preprocessing (impute, encode, scale)
 - optionally runs PCA to reduce dimensionality
 - runs GridSearchCV to pick best k (n_neighbors) and weight/distance options
 - evaluates and saves the best pipeline
"""

import warnings
warnings.filterwarnings("ignore")

import pandas as pd
import numpy as np
from pathlib import Path
import joblib

from sklearn.model_selection import train_test_split, GridSearchCV, StratifiedKFold, cross_val_score
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder, LabelEncoder
from sklearn.neighbors import KNeighborsClassifier
from sklearn.decomposition import PCA
from sklearn.metrics import (
    accuracy_score, precision_score, recall_score, f1_score,
    confusion_matrix, classification_report, roc_auc_score
)

# Optional: for imbalance handling (uncomment to use)
# from imblearn.over_sampling import SMOTE
# from imblearn.pipeline import Pipeline as ImbPipeline

# ---------------------
# Config
# ---------------------
CSV_PATH = "bank-full.csv"
TARGET = "y"
RANDOM_STATE = 42
TEST_SIZE = 0.2
MODEL_OUTPUT = "bank_knn_pipeline.joblib"
USE_PCA = False           # set True to enable PCA before KNN (helpful for high-dim)
PCA_N_COMPONENTS = 20     # only used if USE_PCA = True
DO_SMOTE = False          # set True to apply SMOTE to training set (requires imbalanced-learn)

# ---------------------
# Load dataset
# ---------------------
if not Path(CSV_PATH).exists():
    raise SystemExit(f"Dataset file '{CSV_PATH}' not found in working directory.")

df = pd.read_csv(CSV_PATH, sep=';')  # bank-full.csv uses semicolon delimiter

# Basic cleaning: strip strings
for col in df.select_dtypes(include=["object"]).columns:
    df[col] = df[col].str.strip()

# Remove 'duration' if present (it leaks the label)
if "duration" in df.columns:
    df = df.drop(columns=["duration"])

if TARGET not in df.columns:
    raise SystemExit(f"Target column '{TARGET}' not found. Update TARGET variable as needed.")

# ---------------------
# Features & target
# ---------------------
X = df.drop(columns=[TARGET])
y_raw = df[TARGET].copy()

# Encode target to 0/1 (no=0, yes=1)
le = LabelEncoder()
y = le.fit_transform(y_raw)  # classes order preserved in le.classes_

print("Target classes:", le.classes_)

# ---------------------
# Identify numeric & categorical features
# ---------------------
numeric_features = X.select_dtypes(include=["int64", "float64"]).columns.tolist()
categorical_features = X.select_dtypes(include=["object", "category", "bool"]).columns.tolist()

# Sometimes numeric-looking columns are object due to commas — try coercion
for col in categorical_features[:]:
    coerced = pd.to_numeric(X[col], errors="coerce")
    if coerced.notna().mean() > 0.95:
        X[col] = coerced
        numeric_features.append(col)
        categorical_features.remove(col)

print(f"Numeric features ({len(numeric_features)}): {numeric_features}")
print(f"Categorical features ({len(categorical_features)}): {categorical_features}")

# ---------------------
# Preprocessing pipelines
# ---------------------
numeric_transformer = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="median")),
    ("scaler", StandardScaler())
])

categorical_transformer = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="most_frequent")),
    ("onehot", OneHotEncoder(handle_unknown="ignore", sparse=False))
])

preprocessor = ColumnTransformer(transformers=[
    ("num", numeric_transformer, numeric_features),
    ("cat", categorical_transformer, categorical_features)
], remainder="drop")

# ---------------------
# Build pipeline
# ---------------------
steps = []
steps.append(("preproc", preprocessor))

if USE_PCA:
    steps.append(("pca", PCA(n_components=PCA_N_COMPONENTS, random_state=RANDOM_STATE)))

steps.append(("knn", KNeighborsClassifier()))

pipeline = Pipeline(steps=steps)

# If using SMOTE, create an imblearn pipeline instead (uncomment imports at top)
# if DO_SMOTE:
#     pipeline = ImbPipeline(steps=[("preproc", preprocessor), ("smote", SMOTE(random_state=RANDOM_STATE)), ("knn", KNeighborsClassifier())])

# ---------------------
# Train-test split
# ---------------------
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=TEST_SIZE, stratify=y, random_state=RANDOM_STATE
)

# ---------------------
# Hyperparameter grid for KNN
# ---------------------
param_grid = {
    "knn__n_neighbors": [3, 5, 7, 9, 11, 15],
    "knn__weights": ["uniform", "distance"],
    "knn__p": [1, 2]  # p=1 -> Manhattan, p=2 -> Euclidean
}

# If PCA used, we tuned PCA_N_COMPONENTS manually earlier; otherwise could be added to grid.
cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=RANDOM_STATE)

grid = GridSearchCV(
    pipeline,
    param_grid=param_grid,
    scoring="f1",   # optimize for F1 to balance precision & recall for 'yes' class
    cv=cv,
    n_jobs=-1,
    verbose=1
)

# ---------------------
# Fit grid search
# ---------------------
print("Running GridSearchCV for KNN... (this may take a little while)")
grid.fit(X_train, y_train)

print("Best params:", grid.best_params_)
print("Best CV score (F1):", grid.best_score_)

best_pipeline = grid.best_estimator_

# ---------------------
# Evaluate on test set
# ---------------------
y_pred = best_pipeline.predict(X_test)
y_proba = None
if hasattr(best_pipeline.named_steps["knn"], "predict_proba"):
    try:
        y_proba = best_pipeline.predict_proba(X_test)[:, 1]
    except Exception:
        y_proba = None

acc = accuracy_score(y_test, y_pred)
prec = precision_score(y_test, y_pred, zero_division=0)
rec = recall_score(y_test, y_pred, zero_division=0)
f1 = f1_score(y_test, y_pred, zero_division=0)

print("\nTest evaluation:")
print(f" Accuracy:  {acc:.4f}")
print(f" Precision: {prec:.4f}")
print(f" Recall:    {rec:.4f}")
print(f" F1-score:  {f1:.4f}")
print("\nConfusion Matrix:\n", confusion_matrix(y_test, y_pred))
print("\nClassification Report:\n", classification_report(y_test, y_pred, target_names=le.classes_))

if y_proba is not None:
    try:
        auc = roc_auc_score(y_test, y_proba)
        print(f"ROC AUC: {auc:.4f}")
    except Exception:
        pass

# ---------------------
# Cross-validated performance (optional)
# ---------------------
cv_scores = cross_val_score(best_pipeline, X_train, y_train, cv=cv, scoring="f1", n_jobs=-1)
print("\nCross-validated F1 on training folds:", np.round(cv_scores, 4))
print("Mean CV F1:", np.round(cv_scores.mean(), 4))

# ---------------------
# Save the trained pipeline + label encoder
# ---------------------
joblib.dump({"pipeline": best_pipeline, "label_encoder": le}, MODEL_OUTPUT)
print(f"\nSaved best pipeline + label encoder to: {MODEL_OUTPUT}")

# ---------------------
# Quick predict helper
# ---------------------
def predict_sample(sample_dict):
    """
    sample_dict: dict mapping feature_name -> value (for all features or subset; missing columns will be filled by the pipeline)
    returns predicted label (original string) and probability (if available)
    """
    row = pd.DataFrame([sample_dict])
    pred_code = best_pipeline.predict(row)[0]
    pred_label = le.inverse_transform([pred_code])[0]
    proba = None
    if y_proba is not None:
        try:
            proba = best_pipeline.predict_proba(row)[0][1]  # probability of 'yes'
        except Exception:
            proba = None
    return pred_label, proba
