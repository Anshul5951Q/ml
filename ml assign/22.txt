"""
adult_svm.py

Requirements:
    pip install pandas numpy scikit-learn joblib

Assumptions:
    - Input CSV: 'adult.csv'
    - Target column: 'income' with labels like '>50K' and '<=50K'
    - If your dataset uses different column names, update TARGET and feature lists.
"""

import warnings
warnings.filterwarnings("ignore")

import pandas as pd
import numpy as np
from pathlib import Path
import joblib

from sklearn.model_selection import train_test_split, GridSearchCV, StratifiedKFold
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder, LabelEncoder
from sklearn.svm import SVC, LinearSVC
from sklearn.metrics import (
    accuracy_score, classification_report, confusion_matrix,
    roc_auc_score, roc_curve, precision_recall_fscore_support
)

# -----------------------
# Config
# -----------------------
CSV_PATH = "adult.csv"
TARGET = "income"
RANDOM_STATE = 42
TEST_SIZE = 0.2
MODEL_OUTPUT = "adult_svm_model.joblib"
USE_PROBABILITY = True  # If True, uses SVC(probability=True) (slower). If False, you may use LinearSVC (no predict_proba).

# -----------------------
# Load dataset
# -----------------------
if not Path(CSV_PATH).exists():
    raise SystemExit(f"Dataset file '{CSV_PATH}' not found in working directory.")

df = pd.read_csv(CSV_PATH)

if TARGET not in df.columns:
    raise SystemExit(f"Target column '{TARGET}' not found. Update TARGET variable to match your dataset.")

# Basic cleaning: strip whitespace from string columns (adult dataset often has leading spaces)
for col in df.select_dtypes(include=["object"]).columns:
    df[col] = df[col].str.strip()

# Drop fully-empty columns (if any)
df = df.dropna(axis=1, how="all")

# -----------------------
# Features / target
# -----------------------
X = df.drop(columns=[TARGET])
y_raw = df[TARGET].copy()

# Encode binary labels to 0/1
le = LabelEncoder()
y = le.fit_transform(y_raw)  # e.g., {'<=50K': 0, '>50K': 1}

print("Label classes:", le.classes_)

# -----------------------
# Feature typing (adjust if needed)
# -----------------------
# Common numeric features in Adult dataset
numeric_features = ["age", "fnlwgt", "education_num", "hours_per_week"]
# If your CSV uses slightly different names, update the list above.
# Fallback: infer numeric/categorical
available_cols = X.columns.tolist()
numeric_features = [c for c in numeric_features if c in available_cols]
categorical_features = [c for c in available_cols if c not in numeric_features]

print(f"Numeric features ({len(numeric_features)}): {numeric_features}")
print(f"Categorical features ({len(categorical_features)}): {categorical_features}")

# -----------------------
# Preprocessing pipelines
# -----------------------
numeric_pipeline = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="median")),
    ("scaler", StandardScaler()),
])

categorical_pipeline = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="most_frequent")),
    ("onehot", OneHotEncoder(handle_unknown="ignore", sparse=False))
])

preprocessor = ColumnTransformer(transformers=[
    ("num", numeric_pipeline, numeric_features),
    ("cat", categorical_pipeline, categorical_features)
], remainder="drop")  # drop any unexpected columns

# -----------------------
# Choose estimator (SVC or LinearSVC fallback)
# -----------------------
if USE_PROBABILITY:
    # SVC with probability=True allows ROC AUC and probability outputs (but slower)
    estimator = SVC(probability=True, random_state=RANDOM_STATE)
else:
    # LinearSVC is much faster on large datasets but doesn't support predict_proba.
    estimator = LinearSVC(random_state=RANDOM_STATE, max_iter=5000)

# Complete pipeline
pipeline = Pipeline(steps=[
    ("preproc", preprocessor),
    ("clf", estimator)
])

# -----------------------
# Train / Test split
# -----------------------
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=TEST_SIZE, stratify=y, random_state=RANDOM_STATE
)

# -----------------------
# Hyperparameter grid (example for SVC)
# -----------------------
if isinstance(estimator, SVC):
    param_grid = {
        "clf__C": [0.1, 1, 10],
        "clf__kernel": ["rbf", "linear"],
        "clf__gamma": ["scale", "auto"]  # relevant for rbf
    }
else:
    # LinearSVC grid
    param_grid = {
        "clf__C": [0.01, 0.1, 1, 10],
        "clf__loss": ["squared_hinge"]
    }

cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=RANDOM_STATE)

grid = GridSearchCV(
    pipeline,
    param_grid=param_grid,
    scoring="f1",      # optimize for F1 (balance of precision/recall)
    cv=cv,
    n_jobs=-1,
    verbose=1
)

# -----------------------
# Fit
# -----------------------
print("Starting grid search (this can take a while)...")
grid.fit(X_train, y_train)

print("Best params:", grid.best_params_)
print("Best CV score (F1):", grid.best_score_)

best_model = grid.best_estimator_

# -----------------------
# Evaluate on test set
# -----------------------
y_pred = best_model.predict(X_test)

acc = accuracy_score(y_test, y_pred)
print(f"\nTest Accuracy: {acc:.4f}")

print("\nConfusion Matrix:")
print(confusion_matrix(y_test, y_pred))

print("\nClassification Report:")
print(classification_report(y_test, y_pred, target_names=le.classes_))

# ROC AUC (if probabilities available)
if USE_PROBABILITY and hasattr(best_model.named_steps["clf"], "predict_proba"):
    y_proba = best_model.predict_proba(X_test)[:, 1]
    auc = roc_auc_score(y_test, y_proba)
    print(f"\nROC AUC: {auc:.4f}")
else:
    # fallback: decision_function may be available
    if hasattr(best_model.named_steps["clf"], "decision_function"):
        y_scores = best_model.decision_function(X_test)
        auc = roc_auc_score(y_test, y_scores)
        print(f"\nROC AUC (decision function): {auc:.4f}")

# Precision/Recall/F1 macro
precision, recall, f1, _ = precision_recall_fscore_support(y_test, y_pred, average="binary")
print(f"\nPrecision: {precision:.4f}, Recall: {recall:.4f}, F1: {f1:.4f}")

# -----------------------
# Save model and label encoder
# -----------------------
joblib.dump({"model": best_model, "label_encoder": le}, MODEL_OUTPUT)
print(f"\nSaved trained pipeline + label encoder to: {MODEL_OUTPUT}")

# -----------------------
# Quick predict helper
# -----------------------
def predict_sample(sample_dict):
    """
    sample_dict: dictionary of {feature_name: value} for the dataset features.
    Returns predicted label (original string) and probability (if available).
    """
    row = pd.DataFrame([sample_dict])
    pred_code = best_model.predict(row)[0]
    pred_label = le.inverse_transform([pred_code])[0]
    proba = None
    if USE_PROBABILITY and hasattr(best_model, "predict_proba"):
        proba = best_model.predict_proba(row)[0][pred_code]
    return pred_label, proba
